CREATE TABLE CATEGORIA (
    codigo INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
    );

CREATE TABLE PRODUCTO (
    codigo INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(150),
    precion DECIMAL (10,2) NOT NULL,
    stock INT NOT NULL,
    codigo_categoria INT NOT NULL,
    FOREIGN KEY(codigo_categoria)   REFERENCES CATEGORIA(codigo)
    );
    
CREATE TABLE VENTA (
    numero_venta INT PRIMARY KEY AUTO_INCREMENT,
    fecha VARCHAR(50) NOT NULL,
    dni_cliente INT NULL,
    FOREIGN KEY(dni_cliente) REFERENCE CLIENTE(dni)
    );
    
CREATE TABLE VENTA (
    numero_venta INT PRIMARY KEY AUTO_INCREMENT,
    fecha VARCHAR(50) NOT NULL,
    dni_cliente INT NULL,
    FOREIGN KEY(dni_cliente) REFERENCE CLIENTE(dni)
    );
    
CREATE TABLE DETALLE_VENTA(
    numero_venta INT NOT NULL,
    codigo_producto INT NOT NULL,
    cantidad INT NOT NULL,
    precio_venta DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (numero_venta,codigo_producto),
    FOREIGN KEY (numero_venta) REFERENCES VENTA (numero_venta),
    FOREIGN KEY (codigo_producto) REFERENCES PRODUCTO (codigo)
    );
